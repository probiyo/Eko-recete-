function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); try { var data = JSON.parse(e.postData.contents); // Sütun Sıralaması: Tarih | Bölge | pH | Hava Durumu | Stres/Lab | Notlar | Fotoğraf sheet.appendRow([ data.date || new Date().toLocaleString(), data.region || '', data.ph || '', data.weather || '', data.stress || '', data.notes || '', data.photograph || '' ]); return ContentService.createTextOutput(JSON.stringify({"result": "success"})) .setMimeType(ContentService.MimeType.JSON); } catch(error) { return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": error.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { return ContentService.createTextOutput("Global Agro Doctor API aktif.") .setMimeType(ContentService.MimeType.TEXT); }
